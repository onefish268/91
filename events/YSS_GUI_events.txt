	#检测最低限度20%
	id = YSS.1000
	is_triggered_only = yes
	hidden = yes
	immediate = {
		if = {
			limit = {
				has_country_flag = {
					flag = YSS_prepare_support
					days > 29
				}
			}

			if = {
				limit = {
					communism > 0.2
				}
				YSS = {
					country_event = YSS.1001
				}
			}
			else = {
				set_country_flag = {
					flag = YSS_prepare_support
					days = 31
					value = 1
				}

				country_event = {
					id = YSS.1000
					days = 30
				}
			}
		}
	}
