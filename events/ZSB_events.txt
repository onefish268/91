    id = ZSB.15
    hidden = yes
    is_triggered_only = yes

    immediate = {
        
        if = {
            limit = { NOT = { has_country_flag = ZSB_the_vote_has_over } }

            random_scope_in_array = {#从数组中选取一个地区 斯坦莉的
                array = ZSB_all_regions
                PREV = { set_temp_variable = { ZSB_random_region_0 = PREV } }
            }

            remove_from_array = {#去除原有的地区（防止连续重复发生同一地区决议）
				array = ZSB_all_regions
				value = ZSB_random_region_0
			}

            random_list = {
                10 = {
                    modifier = {
			    		factor = 0
			    		NOT = { check_variable = { ZSB_decision_02_var = 0 } }
	    			}
                    set_variable = { ZSB_decision_02_var = ZSB_random_region_0 }
                    activate_mission = ZSB_decision_02
                }
                10 = {
                    modifier = {
						factor = 0
						NOT = { check_variable = { ZSB_decision_04_var = 0 } }
    				}

                    activate_mission = ZSB_decision_04
