country_event = {#关于入侵南非的隐藏触发器
	id = invisible_event.007
	is_triggered_only = yes
	hidden = yes

	trigger = {
		ROOT = {
			has_war_with = EAS
		}
	}

	immediate = {
		random_country = {
		    limit = {
		    	has_country_flag = the_lord_of_EAS
		    }
		    set_autonomy = {
		        target = EAS
	      	    autonomous_state = autonomy_colony
	    	    end_wars = no
		    	end_civil_wars = no
		   	}
		}
	}
}
