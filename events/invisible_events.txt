country_event = {#配帕洛尼自动走国策
	id = invisible_event.002
	is_triggered_only = yes
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_completed_focus = 3_focus_Endless_fight
			}
		}
		else_if = {
			limit = {
				has_completed_focus = three_focus_22
			}
			complete_national_focus = {
				focus = 3_focus_Endless_fight
				use_side_message = yes
				originator_name = "配帕洛尼"
			}
			deactivate_shine_on_focus = 3_focus_Endless_fight
			clr_country_flag = seven_time
		}
		else_if = {
			limit = {
				has_completed_focus = three_focus_17
			}
			complete_national_focus = {
				focus = three_focus_22
				use_side_message = yes
				originator_name = "配帕洛尼"
			}
			deactivate_shine_on_focus = three_focus_22
			clr_country_flag = seven_time
		}
		else = {
			complete_national_focus = {
				focus = three_focus_17
				use_side_message = yes
				originator_name = "配帕洛尼"
			}
			activate_shine_on_focus = three_focus_22
			deactivate_shine_on_focus = three_focus_17
			country_event = {
				id = invisible_event.002
				days = 140
			}
			set_country_flag = {
				flag = seven_time
				days = 141
				value = 1
			}
		}
	}
}
