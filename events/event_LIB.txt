country_event = {
	id = LIB_lw.2
	title = LIB_lw.2.t
	desc = LIB_lw.2.d
	fire_only_once = yes
	is_triggered_only = yes
	immediate = {
        set_global_flag = FRA_has_fallen
        if = {
            limit = {
                # 检查ROOT（KMK）的领袖是否是蕾米
                KMK = {
                    has_country_leader = {
                        character = KMK_Remilia
                        ruling_only = yes
                    }
                }
            }
            LIB = {
                set_country_flag = LIB_remi_in_power_flag
            }
        }
		else = {
			LIB = {
                set_country_flag = LIB_fran_in_power_flag
            }
		}
    }
	option = {
		name = LIB_lw.2.a
		KMK = {
			transfer_navy = {
				target = LIB
			}
		}
		if = {
			limit = {
				is_faction_leader  = yes
			}
			dismantle_faction = yes
		}
		if = {
			limit = {
				LIB = {
					NOT = {
						has_war_with = DVL
					}
				}
			}
			DVL = {
				declare_war_on = { target = LIB type = take_state }
			}
		}
		LIB = {
			set_major = yes
			
			country_event = {
				id = LIB_lw.3
				days = 1
			}
			KMK = {
				set_autonomy = {
					target = LIB
					autonomy_state = autonomy_free
				}
			}
			add_ideas = LIB_never_surrunder
			change_tag_from_ROOT = yes
			drop_cosmetic_tag = yes
			set_cosmetic_tag = LIB_FRA
		}
		hidden_effect = {
			
			KMK = {
				diplomatic_relation = {
					country = RIS
					relation = guarantee
					active = no
				}
			}
			LIB = {
				diplomatic_relation = {
					country = RIS
					relation = guarantee
					active = yes
				}
			}
		}	
		ai_chance = {
			base = 10
		}
	}
}
