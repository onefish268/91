ZSB_count_STL_number = {
	while_loop_effect = {
		limit = { check_variable = { ZSB_STL_ones_number > 10 } }
		add_to_variable = { ZSB_STL_ones_number = -10 }
		add_to_variable = { ZSB_STL_tens_number = 1 }
	}
	if = {
		limit = { check_variable = { ZSB_STL_hundreds_number > 1 } }
		set_variable = { ZSB_STL_ones_number = 1 }
		set_variable = { ZSB_STL_tens_number = 1 }
		set_variable = { ZSB_STL_hundreds_number = 2 }
	}
	else_if = {
		limit = { check_variable = { ZSB_STL_tens_number > 10 }  }
		set_variable = { ZSB_STL_ones_number = 1 }
		set_variable = { ZSB_STL_tens_number = 1 }
		set_variable = { ZSB_STL_hundreds_number = 2 }
	}
}
