YSS_return_button_effect_0 = {
	#获得争对某个国家的buff
	if = {
		limit = {
			var:YSS_chosen_array^0 = {
				NOT = { has_country_flag = YSS_revoluntion_victory }
				NOT = { has_idea = the_influence_of_YSS }
			}
			has_political_power > 49
		}
		add_political_power = -50

		var:YSS_chosen_array^0 = {
			add_timed_idea = {
				idea = the_influence_of_YSS

			}

		}
	}

YSS_return_button_effect_2 = {
	#设置外派的将领
	if = {
		limit = {
			var:YSS_chosen_array^0 = { NOT = { has_country_flag = YSS_revoluntion_victory } }
			any_army_leader = {
				NOT = { has_country_flag = YSS_has_mission_to_aboard }
			}
		}

		meta_effect = {
			text = {
				add_to_variable = { YSS.YSS_[country_tag]_willing_join = 2 }
			}
			country_tag = "[?YSS.YSS_chosen_array^0.GetTag]"
		}

			if = {
				limit = { var:YSS_chosen_array^0 = { has_country_flag = YSS_launch_the_civil_war } }

				random_army_leader = {
					limit = { NOT = { has_character_flag = YSS_has_mission_to_aboard } }
					set_character_flag = YSS_has_mission_to_aboard
					meta_effect = {
						text = {
							set_character_flag = go_to_[country_tag]
							set_nationality = [country_tag]
						}
						debug = yes
						country_tag = "[?YSS.YSS_chosen_array^0.GetTag]"
					}
				}

				var:YSS_chosen_array^0 = {
					add_to_variable = { YSS_officer_factor = 0.05 }
				}
			}
			else = {
				random_army_leader = {
					limit = { NOT = { has_character_flag = YSS_has_mission_to_aboard } }
					set_character_flag = YSS_has_mission_to_aboard
					meta_effect = {
						text = {
							set_character_flag = go_to_[country_tag]
						}
						country_tag = "[?YSS.YSS_chosen_array^0.GetTag]"
					}
				}
			}
	}
}

YSS_return_button_effect_4 = {#给工厂
	#给军工
	if = {
		limit = {
			var:YSS_chosen_array^0 = { NOT = { has_country_flag = YSS_revoluntion_victory } }
			
			check_expr = {
        		value = YSS.num_of_military_factories
				add = YSS.YSS_the_num_of_send_factories
        		subtract = YSS.YSS_total_of_factory_COG
				subtract = YSS.YSS_total_of_factory_ZIM
				subtract = YSS.YSS_total_of_factory_ZYX
				subtract = YSS.YSS_total_of_factory_ZSB
				subtract = YSS.YSS_total_of_factory_EAS
				subtract = YSS.YSS_total_of_factory_EGY
				subtract = YSS.YSS_total_of_factory_LBA
				subtract = YSS.YSS_total_of_factory_LIB
				subtract = YSS.YSS_total_of_factory_SAF
        		greater_than = 1
    		}
		}

		meta_effect = {
			text = {
				add_to_variable = { YSS.YSS_[country_tag]_willing_join = 5 }
			}
			country_tag = "[?YSS.YSS_chosen_array^0.GetTag]"
		}

		if = {
			limit = { var:YSS.YSS_chosen_array^0 = { has_country_flag = YSS_launch_the_civil_war } }

			meta_effect = {
					text = {
						add_to_variable = { YSS.YSS_total_of_factory_[country_tag] = 2 }
					}
					country_tag = "[?var:YSS.YSS_chosen_array^0.GetTag]"
				}

						var:YSS.YSS_chosen_array^0 = {
							add_offsite_building = {
		            			type = arms_factory
            					level = 2
        					}
						}

						set_temp_variable = { round_i = 0 }

						while_loop_effect = {
							limit = {
								check_variable = { round_i < 2 }
							}

							add_to_temp_variable = { round_i = 1 }

							YSS = {
								random_owned_controlled_state = {
    								limit = {
        								arms_factory > 0
			    					}

								    remove_building = {
										type = arms_factory
										level = 1
									}
								}
								add_to_variable = { YSS.YSS_the_num_of_send_factories = 1 }
							}
						}
		}
		else = {
			meta_effect = {
				text = {
					add_to_variable = { YSS.YSS_total_of_factory_[country_tag] = 2 }
				}
				country_tag = "[?var:YSS.YSS_chosen_array^0.GetTag]"
			}
		}
	}
}
