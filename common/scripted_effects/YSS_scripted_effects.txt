YSS_upgrade_state_admin = {
    if = {
        limit = { has_dynamic_modifier = { modifier = YSS_province_xingzheng_1 } }
        remove_dynamic_modifier = { modifier = YSS_province_xingzheng_1 }
        add_dynamic_modifier = { modifier = YSS_province_xingzheng_2 }
    }
    else_if = {
        limit = { has_dynamic_modifier = { modifier = YSS_province_xingzheng_2 } }
        remove_dynamic_modifier = { modifier = YSS_province_xingzheng_2 }
        add_dynamic_modifier = { modifier = YSS_province_xingzheng_3 }
    }
    else_if = {
        limit = { has_dynamic_modifier = { modifier = YSS_province_xingzheng_3 } }
        remove_dynamic_modifier = { modifier = YSS_province_xingzheng_3 }
        add_dynamic_modifier = { modifier = YSS_province_xingzheng_4 }
    }
    else_if = {
        limit = { has_dynamic_modifier = { modifier = YSS_province_xingzheng_4 } }
        remove_dynamic_modifier = { modifier = YSS_province_xingzheng_4 }
        add_dynamic_modifier = { modifier = YSS_province_xingzheng_5 }
    }
}

YSS_set_pubilcity_slot = {
	if = {
		limit = {
			check_variable = {
				YSS_pub_slot_A = 0
			}
		}
		set_variable = {
			YSS_pub_slot_A = YSS_pub_temp
		}
		else_if = {
			limit = {
				check_variable = {
					YSS_pub_slot_B = 0
				}
			}
			set_variable = {
				YSS_pub_slot_B = YSS_pub_temp
			}
			else_if = {
				limit = {
					check_variable = {
						YSS_pub_slot_C = 0
					}
				}
				set_variable = {
					YSS_pub_slot_C = YSS_pub_temp
				}
			}
		}
	}
}
