YSS_return_button_effect_0 = {
	#获得争对某个国家的buff
	if = {
		limit = {
			has_political_power > 49
			all_of_scopes = {
				array = YSS_chosen_array
				
				NOT = { has_country_flag = YSS_revoluntion_victory }
			}
		}
		add_political_power = -50
		for_each_scope_loop = {
			array = YSS_chosen_array
			add_timed_idea = {
				idea = the_influence_of_YSS
				days = 365
			}
			if = {
				limit = {
					has_country_flag = YSS_we_start_to_action
				}
			}
			else = {
				country_event = YSS.1000
			}
		}
	}
}

YSS_return_button_effect_2 = {
	#设置外派的将领
	if = {
		limit = {
			all_of_scopes = {
				array = YSS_chosen_array
				
				NOT = { has_country_flag = YSS_revoluntion_victory }
			}
		}
		for_each_scope_loop = {
			array = YSS_chosen_array
			if = {
				#战争已经开打
				limit = {
					has_country_flag = YSS_launch_the_civil_war
					has_war = yes
				}
				YSS = {
					random_army_leader = {
						limit = {
							NOT = {
								has_character_flag = YSS_has_mission_to_aboard
							}
						}
						set_character_flag = YSS_has_mission_to_aboard
						meta_effect = {
							text = {
								set_character_flag = go_to_[country_tag]
							}
							country_tag = "[?var:YSS_chosen_array^0]"
						}
						set_nationality = PREV.PREV
					}
				}
			}
			else_if = {
				limit = {
					NOT = {
						has_country_flag = YSS_launch_the_civil_war
					}
				}
				#战事未开
				YSS = {
					random_army_leader = {
						limit = {
							NOT = {
								has_character_flag = YSS_has_mission_to_aboard
							}
						}
						set_character_flag = YSS_has_mission_to_aboard
						meta_effect = {
							text = {
								set_character_flag = go_to_[country_tag]
							}
							country_tag = "[?var:YSS_chosen_array^0]"
						}
					}
				}
			}
		}
	}
}

YSS_return_button_effect_3 = {
	if = {
		limit = {
			all_of_scopes = {
				array = YSS_chosen_array
				
				NOT = { has_country_flag = YSS_revoluntion_victory }
			}
		}
		#增加志愿上限
		for_each_scope_loop = {
			array = YSS_chosen_array
			if = {
				#没有设置过flag
				limit = {
					NOT = {
						has_country_flag = YSS_we_have_volunteer_to_this_country
					}
					has_war = yes
					has_political_power > 49
				}
				add_political_power = -50
				set_country_flag = YSS_we_have_volunteer_to_this_country
				add_to_variable = {
					YSS_volunteer_to_other_country_send_volunteer_size = 1
				}
			}
			else_if = {
				limit = {
					NOT = {
						has_country_flag = YSS_launch_the_civil_war
					}
					NOT = {
						has_country_flag = YSS_we_prepare_to_send_volunteer
					}
					has_political_power > 49
				}
				add_political_power = -50
				set_country_flag = YSS_we_prepare_to_send_volunteer
			}
		}
	}
}

YSS_return_button_effect_4 = {
	#给军工
	if = {
		limit = {
			all_of_scopes = {
				array = YSS_chosen_array
				
				NOT = { has_country_flag = YSS_revoluntion_victory }
			}
			num_of_military_factories > 1
		}
		set_temp_variable = {
			i = 0
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					i < 2
				}
			}
			add_to_temp_variable = {
				i = 1
			}
			random_controlled_state = {
				limit = {
					check_variable = {
						building_level@arms_factory > 0
					}
				}
				add_building_construction = {
					type = arms_factory
					level = -1
					instant_build = yes
				}
			}
		}
		for_each_scope_loop = {
			array = YSS_chosen_array
			if = {
				limit = {
					has_country_flag = YSS_launch_the_civil_war
				}
				while_loop_effect = {
					limit = {
						check_variable = {
							i > 0
						}
					}
					add_to_temp_variable = {
						i = -1
					}
					random_controlled_state = {
						limit = {
							free_building_slots = {
								building = arms_factory
								size > 0
							}
						}
						add_building_construction = {
							type = arms_factory
							level = 1
							instant_build = yes
						}
					}
				}
			}
			else = {
				add_to_variable = { YSS_total_of_factory = 2 }
			}
		}
	}
}
