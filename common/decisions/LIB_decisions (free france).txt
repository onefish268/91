#隙间民工
LIB_xijian_mg = {
    icon = GFX_decision_generic_political_actions
    cost = 0
    days_mission_timeout = 10
    fire_only_once = no
	available = {
        always = no
    }
	activation = {always = no}
    complete_effect = {
        # 循环继续
    }

    timeout_effect = {
        # 给一个军工或地图外军工
		random_state = {
            limit = {
			OR = {
                is_core_of = FRA
				is_core_of = KMK
				is_core_of = KNM
				}
			industrial_complex > 0
            }
            remove_building = {
                type = industrial_complex
                level = 1
            }
        }
        if = {
            limit = {
                any_state = {
                    is_core_of = ROOT
                   free_building_slots = { building = industrial_complex size > 0 include_locked = no }
                }
            }
            random_state = {
                limit = {
                    is_core_of = ROOT
                    free_building_slots = { building = industrial_complex size > 0 include_locked = no }
                }
                add_building_construction = {
                    type = industrial_complex
                    level = 1
                    instant_build = yes
                }
            }
        }
        else = {
            add_offsite_building = {
                type = industrial_complex
                level = 1
            }
            add_to_variable = {
                var = LIB_xijianrongliang
                value = 1
            }
            add_to_variable = {
                var = LIB_xijianrongliang_mg
                value = 1
            }
        }

        # 判断是否继续循环
        if = {
            limit = {
			has_completed_focus = LIB_new_focus_43
                check_variable = { LIB_xijianrongliang < LIB_xijianshangxian }
			any_state = {
				OR = {
                    is_core_of = FRA
					is_core_of = KMK
					is_core_of = KNM
					}
                industrial_complex > 0
				}
                }
            activate_mission = LIB_xijian_mg

            # 若已完成特定国策，缩短下次任务时长
            if = {
                limit = {
					has_completed_focus = LIB_new_focus_43
                    has_completed_focus = LIB_new_focus_44
                }
                add_days_mission_timeout = {
                    mission = LIB_xijian_mg
                    days = -2
                }
            }
        }
        else = {
            clr_country_flag = LIB_loop_active_mg
            add_to_variable = {
                var = LIB_civilian_factory_use
                value = -5
                tooltip = civilian_factory_use_tt
            }
        }
    }
}
