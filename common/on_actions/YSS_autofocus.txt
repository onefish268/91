on_actions = {
	# ── YSS「大庆喜」党纪国策链：自动推进，玩家无需手动点 ──
	# 每完成一环，下一环在 70 天后自动完成（相当于默认国策时长）
	on_daily = {
		effect = {
			if = {
				limit = { tag = YSS }
